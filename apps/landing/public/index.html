Quickstart 
const ticket = await getOpenTicketInstructionAsync({
  ...accounts,
  price: 1_000_000n, // 1 USDC
  payoutTable: [
    { p: 96_500_000n, m: 100_000n }, // 10x, 9.65%
    { p: 903_500_000n, m: 0n },
  ],
  appShareBps: 0n,
  nonce, // the app's next ticket number
});

// One signature: the payment and its ticket.
await send(user, [payment, ticket]);
